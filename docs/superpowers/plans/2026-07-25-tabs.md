# Tabs / Multi-Document Editing Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Open multiple documents as tabs in the center workspace, one visible at a time, closing issue #20's tabs half and unblocking the CLI command console.

**Architecture:** A pure, immutable `TabSet` value (same style as the existing `navHistory.ts`) replaces `App.tsx`'s `view` / `selectedPath` / `navState` signals. Autosave stays global and unchanged; activating a tab flushes it first, which guarantees only the active tab is ever dirty. Because an evicted tab is therefore always clean, live `Editor` instances are capped by an LRU and evicted tabs rehydrate from disk.

**Tech Stack:** Solid + TypeScript (`ui/`), Vitest, Rust + Tauri (`crates/cubical-app`), `serde_json`.

**Spec:** [`../specs/2026-07-25-tabs-design.md`](../specs/2026-07-25-tabs-design.md)

## Global Constraints

- **No explanatory comments in source** — none at all, doc-comments (`///`, `//!`, JSDoc) included. A brief one-liner is the maximum. Rationale goes in `docs/implementation/frontend.md`. See `docs/conventions.md` → Comments for the functional pragmas that must never be stripped.
- **Splits are out of scope.** Tabs only — no pane tree, no focus model, no per-pane strips.
- **No duplicate tabs of one document.** `id` is derived from the view, so opening an already-open document activates its existing tab.
- **Only the active tab can be dirty.** Activation flushes autosave first. Never introduce per-tab `dirty` / timer / `pendingWrite`.
- **Autosave machinery stays global and unchanged** — `dirty`, `autosaveTimer`, `pendingWrite`, `seenHash`, `lastWrittenHash` remain single module-level values in `App.tsx`.
- **Evicted tabs rehydrate from disk**, never from an in-memory content cache. Plain `.md` files are the source of truth.
- **`editor.live_tab_limit`** — number, default `8`, clamped to a minimum of `1`.
- **Tab sessions are machine-local** — a JSON file in the Tauri app-data dir, never `.cubical/config.toml`.
- **All IPC goes through `ui/src/api/ipc.ts`**, the single frontend chokepoint (`docs/migration-touchpoints.md`).
- **Full gate is `scripts/check.sh`.** Never pipe it to `tail` — that reports `tail`'s exit code, not the gate's. Redirect to a file and check `$?`.
- **Known flake:** `cubical-core`'s `watcher::…dropping_handle_stops_event_delivery_within_100ms` fails under full-workspace load and passes in isolation. Do not "fix" it.

```ts
import { emptyNav, type NavState } from "../navHistory";

export type TabView =
  | { kind: "file"; path: string }
  | { kind: "tag"; tagPath: string };

export interface Tab {
  id: string;
  view: TabView;
  nav: NavState;
}

export interface TabSet {
  tabs: Tab[];
  activeId: string | null;
}

export const emptyTabs: TabSet = { tabs: [], activeId: null };

export function tabId(view: TabView): string {
  return view.kind === "file" ? `file:${view.path}` : `tag:${view.tagPath}`;
}

export function activeTab(s: TabSet): Tab | null {
  return s.tabs.find((t) => t.id === s.activeId) ?? null;
}

export function openTab(s: TabSet, view: TabView): TabSet {
  const id = tabId(view);
  if (s.tabs.some((t) => t.id === id)) return { ...s, activeId: id };
  return { tabs: [...s.tabs, { id, view, nav: emptyNav }], activeId: id };
}

export function closeTab(s: TabSet, id: string): TabSet {
  const i = s.tabs.findIndex((t) => t.id === id);
  if (i < 0) return s;
  const tabs = s.tabs.filter((t) => t.id !== id);
  if (s.activeId !== id) return { ...s, tabs };
  const next = tabs[i] ?? tabs[i - 1] ?? null;
  return { tabs, activeId: next ? next.id : null };
}

export function activateTab(s: TabSet, id: string): TabSet {
  return s.tabs.some((t) => t.id === id) ? { ...s, activeId: id } : s;
}

export function moveTab(s: TabSet, id: string, toIndex: number): TabSet {
  const from = s.tabs.findIndex((t) => t.id === id);
  if (from < 0) return s;
  const clamped = Math.max(0, Math.min(s.tabs.length - 1, toIndex));
  const tabs = [...s.tabs];
  const [moved] = tabs.splice(from, 1);
  tabs.splice(clamped, 0, moved);
  return { ...s, tabs };
}

function step(s: TabSet, delta: number): TabSet {
  if (s.tabs.length === 0) return s;
  const i = s.tabs.findIndex((t) => t.id === s.activeId);
  const base = i < 0 ? 0 : i;
  const n = s.tabs.length;
  return { ...s, activeId: s.tabs[(base + delta + n) % n].id };
}

export function nextTab(s: TabSet): TabSet {
  return step(s, 1);
}

export function prevTab(s: TabSet): TabSet {
  return step(s, -1);
}

export function updateNav(
  s: TabSet,
  id: string,
  fn: (nav: NavState) => NavState,
): TabSet {
  return {
    ...s,
    tabs: s.tabs.map((t) => (t.id === id ? { ...t, nav: fn(t.nav) } : t)),
  };
}

export function remapTabPaths(
  s: TabSet,
  remap: (path: string) => string | null,
): TabSet {
  const seen = new Set<string>();
  const tabs: Tab[] = [];
  const rename = new Map<string, string>();
  for (const t of s.tabs) {
    let next = t;
    if (t.view.kind === "file") {
      const to = remap(t.view.path);
      if (to !== null && to !== t.view.path) {
        const view: TabView = { kind: "file", path: to };
        next = { ...t, id: tabId(view), view };
        rename.set(t.id, next.id);
      }
    }
    if (seen.has(next.id)) continue;
    seen.add(next.id);
    tabs.push(next);
  }
  const activeId = s.activeId === null ? null : (rename.get(s.activeId) ?? s.activeId);
  return { tabs, activeId: tabs.some((t) => t.id === activeId) ? activeId : (tabs[0]?.id ?? null) };
}

export function dropMissingTabs(
  s: TabSet,
  exists: (path: string) => boolean,
): TabSet {
  const tabs = s.tabs.filter((t) => t.view.kind !== "file" || exists(t.view.path));
  if (tabs.length === s.tabs.length) return s;
  const activeId = tabs.some((t) => t.id === s.activeId)
    ? s.activeId
    : (tabs[0]?.id ?? null);
  return { tabs, activeId };
}
```

```ts
export const DEFAULT_LIVE_TAB_LIMIT = 8;

export function clampLimit(raw: number): number {
  if (!Number.isFinite(raw)) return DEFAULT_LIVE_TAB_LIMIT;
  return Math.max(1, Math.floor(raw));
}

export function touch(mru: string[], id: string): string[] {
  return [id, ...mru.filter((x) => x !== id)];
}

export function liveIds(
  mru: string[],
  activeId: string | null,
  limit: number,
): string[] {
  const cap = clampLimit(limit);
  const ordered = activeId === null ? mru : touch(mru, activeId);
  return ordered.slice(0, cap);
}
```

```rust
type Store = BTreeMap<String, TabSession>;

fn read_all(store: &Path) -> Store {
    match std::fs::read(store) {
        Ok(bytes) => serde_json::from_slice(&bytes).unwrap_or_default(),
        Err(_) => Store::default(),
    }
}

pub fn load(store: &Path, vault_path: &str) -> TabSession {
    read_all(store).remove(vault_path).unwrap_or_default()
}

pub fn save(store: &Path, vault_path: &str, session: &TabSession) {
    let mut all = read_all(store);
    if session.tabs.is_empty() {
        all.remove(vault_path);
    } else {
        all.insert(vault_path.to_string(), session.clone());
    }
    let _ = atomic_write(store, &all);
}

fn atomic_write(store: &Path, all: &Store) -> std::io::Result<()> {
    if let Some(parent) = store.parent() {
        std::fs::create_dir_all(parent)?;
    }
    let json = serde_json::to_vec_pretty(all)?;
    let tmp = store.with_extension("json.tmp");
    std::fs::write(&tmp, &json)?;
    std::fs::rename(&tmp, store)
}
```

```tsx
import { For } from "solid-js";
import type { Tab, TabSet } from "./tabModel";

export interface TabStripProps {
  tabs: TabSet;
  onActivate: (id: string) => void;
  onClose: (id: string) => void;
  onMove: (id: string, toIndex: number) => void;

function label(tab: Tab): string {
  if (tab.view.kind === "tag") return `#${tab.view.tagPath}`;
  const base = tab.view.path.slice(tab.view.path.lastIndexOf("/") + 1);
  return base.endsWith(".md") ? base.slice(0, -3) : base;

export default function TabStrip(props: TabStripProps) {

      <For each={props.tabs.tabs}>
        {(tab, index) => (
          <div
            class="tab"
            classList={{ "tab--active": tab.id === props.tabs.activeId }}
            role="tab"
            aria-selected={tab.id === props.tabs.activeId}
            draggable
            onClick={() => props.onActivate(tab.id)}
            onDragStart={() => {
              dragged = tab.id;
            }}
            onDragOver={(e) => e.preventDefault()}
            onDrop={() => {
              if (dragged !== null && dragged !== tab.id) {
                props.onMove(dragged, index());
              }
              dragged = null;
            }}

            <button
              class="tab__close"
              aria-label="Close tab"
              onClick={(e) => {
                e.stopPropagation();
                props.onClose(tab.id);
              }}
            >
              ×
            </button>
          </div>
        )}
      </For>
```

```ts
export interface ActivationDeps {
  current: () => TabSet;
  flush: () => Promise<void>;

  loadContent: () => Promise<void>;

export async function activateWithFlush(
  deps: ActivationDeps,
  id: string,
): Promise<void> {
  if (deps.current().activeId === id) return;
  await deps.flush();
  deps.resetDocState();
```

```tsx
const toDto = (s: TabSet): TabSessionDto => ({
  active_id: s.activeId,
  tabs: s.tabs.map((t) => ({
    id: t.id,
    kind: t.view.kind,
    path: t.view.kind === "file" ? t.view.path : null,
    tag_path: t.view.kind === "tag" ? t.view.tagPath : null,
  })),

const fromDto = (dto: TabSessionDto): TabSet => {
  const tabs = dto.tabs.flatMap((r) => {
    const view: TabView | null =
      r.kind === "file" && r.path !== null
        ? { kind: "file", path: r.path }
        : r.kind === "tag" && r.tag_path !== null
          ? { kind: "tag", tagPath: r.tag_path }
          : null;
    return view === null ? [] : [{ id: tabId(view), view, nav: emptyNav }];
  });
  const activeId = tabs.some((t) => t.id === dto.active_id)
    ? dto.active_id
    : (tabs[0]?.id ?? null);
  return { tabs, activeId };
};
```
